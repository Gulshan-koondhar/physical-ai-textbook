TensorRT Optimization Examples for Jetson Orin

Introduction

NVIDIA TensorRT is a high-performance deep learning inference optimizer and runtime that delivers low latency and high throughput for AI applications. For humanoid robots deployed on NVIDIA Jetson Orin platforms, TensorRT optimization is crucial for achieving real-time performance while maintaining energy efficiency. This chapter provides comprehensive examples and best practices for optimizing AI models using TensorRT specifically for Jetson Orin deployment.

The Jetson AGX Orin platform features powerful AI accelerators including the Deep Learning Accelerator (DLA) and programmable vision accelerators (PVA), which can be leveraged through TensorRT optimization to achieve optimal performance for humanoid robot applications.

Understanding TensorRT Architecture

TensorRT Components

TensorRT consists of several key components:

  1. Parser: Converts trained models from various frameworks (ONNX, TensorFlow, PyTorch)
  2. Builder: Creates optimized inference engines
  3. Runtime: Executes inference using optimized engines
  4. Optimizer: Applies various optimization techniques

Optimization Techniques

TensorRT applies several optimization techniques:

  • Kernel Fusion: Combines operations to reduce memory access
  • Precision Calibration: Supports INT8 and FP16 precision
  • Layer Fusion: Fuses compatible layers
  • Memory Optimization: Optimizes memory allocation
  • Dynamic Tensor Memory: Efficiently manages temporary tensors

Simple Model Optimization

# basic_tensorrt_optimization.py
import tensorrt as trt
import pycuda.driver as cuda
import pycuda.autoinit
import numpy as np
import torch
import torch.nn as nn

class SimpleModel(nn.Module):
    def __init__(self):
        super(SimpleModel, self).__init__()
        self.conv1 = nn.Conv2d(3, 64, 3, padding=1)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool2d(2)
        self.conv2 = nn.Conv2d(64, 128, 3, padding=1)
        self.fc1 = nn.Linear(128 * 14 * 14, 256)  # Assuming 28x28 input
        self.fc2 = nn.Linear(256, 10)

    def forward(self, x):
        x = self.relu(self.conv1(x))
        x = self.pool(x)
        x = self.relu(self.conv2(x))
        x = self.pool(x)
        x = x.view(x.size(0), -1)
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

def create_tensorrt_engine(model, input_shape, precision="fp16"):
    """
    Create TensorRT engine from PyTorch model

    Args:
        model: PyTorch model
        input_shape: Input tensor shape [batch, channels, height, width]
        precision: Precision mode ("fp32", "fp16", "int8")

    Returns:
        TensorRT engine
    """
    logger = trt.Logger(trt.Logger.WARNING)

    # Create builder
    builder = trt.Builder(logger)

    # Create network
    network_flags = 1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    network = builder.create_network(flags=network_flags)

    # Create config
    config = builder.create_builder_config()

    # Set memory limits appropriate for Jetson
    config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE, 2 << 30)  # 2GB

    # Create optimization profile
    profile = builder.create_optimization_profile()

    # Define input dimensions
    input_name = "input"
    input_dims = trt.Dims4(input_shape)  # [N, C, H, W]

    # Set profile shapes (min, opt, max)
    profile.set_shape(input_name,
                     min=(1, input_shape[1], input_shape[2], input_shape[3]),  # Min batch
                     opt=(1, input_shape[1], input_shape[2], input_shape[3]),  # Optimal batch
                     max=(4, input_shape[1], input_shape[2], input_shape[3]))  # Max batch
    config.add_optimization_profile(profile)

    # Convert PyTorch model to ONNX first
    dummy_input = torch.randn(input_shape)
    onnx_path = "temp_model.onnx"

    torch.onnx.export(
        model,
        dummy_input,
        onnx_path,
        export_params=True,
        opset_version=11,
        do_constant_folding=True,
        input_names=[input_name],
        output_names=["output"],
        dynamic_axes={
            input_name: {0: "batch_size"},
            "output": {0: "batch_size"}
        }
    )

    # Parse ONNX model
    parser = trt.OnnxParser(network, logger)
    with open(onnx_path, 'rb') as model_file:
        if not parser.parse(model_file.read()):
            print("ERROR: Failed to parse the ONNX file.")
            for error in range(parser.num_errors):
                print(parser.get_error(error))
            return None

    # Set precision
    if precision == "fp16":
        if builder.platform_has_fast_fp16:
            config.set_flag(trt.BuilderFlag.FP16)
        else:
            print("FP16 not supported on this platform")

    elif precision == "int8":
        if builder.platform_has_fast_int8:
            config.set_flag(trt.BuilderFlag.INT8)
            # For INT8, you need calibration data
            # config.int8_calibrator = create_calibrator(calibration_data)
        else:
            print("INT8 not supported on this platform")

    # Build engine
    print("Building TensorRT engine...")
    engine = builder.build_engine(network, config)

    if engine is None:
        print("Failed to build engine")
        return None

    print(f"Engine built successfully with precision: {precision}")
    return engine

def run_inference_with_engine(engine, input_data):
    """Run inference using TensorRT engine"""
    # Create runtime
    runtime = trt.Runtime(trt.Logger(trt.Logger.WARNING))

    # Deserialize engine if needed
    if isinstance(engine, bytes):
        engine = runtime.deserialize_cuda_engine(engine)

    # Create execution context
    context = engine.create_execution_context()

    # Allocate buffers
    inputs, outputs, bindings, stream = allocate_buffers(engine, input_data.shape)

    # Copy input data to GPU
    np.copyto(inputs[0]['host'], input_data.ravel())
    cuda.memcpy_htod(inputs[0]['device'], inputs[0]['host'])

    # Run inference
    context.execute_async_v2(bindings=bindings, stream_handle=stream.handle)

    # Copy output data back to CPU
    cuda.memcpy_dtoh(outputs[0]['host'], outputs[0]['device'])

    # Synchronize stream
    stream.synchronize()

    return outputs[0]['host']

def allocate_buffers(engine, input_shape):
    """Allocate input and output buffers for inference"""
    inputs = []
    outputs = []
    bindings = []
    stream = cuda.Stream()

    for idx in range(engine.num_bindings):
        binding_name = engine.get_binding_name(idx)
        binding_shape = engine.get_binding_shape(idx)

        if binding_shape[0] == -1:  # Dynamic batch dimension
            binding_shape = tuple([input_shape[0]] + list(binding_shape[1:]))

        size = trt.volume(binding_shape)
        dtype = trt.nptype(engine.get_binding_dtype(idx))

        # Allocate host and device buffers
        host_mem = cuda.pagelocked_empty(size, dtype)
        device_mem = cuda.mem_alloc(host_mem.nbytes)

        bindings.append(int(device_mem))

        if engine.binding_is_input(idx):
            inputs.append({'host': host_mem, 'device': device_mem})
        else:
            outputs.append({'host': host_mem, 'device': device_mem})

    return inputs, outputs, bindings, stream

def main():
    # Create and prepare model
    model = SimpleModel()
    model.eval()

    # Input shape: [batch, channels, height, width]
    input_shape = (1, 3, 28, 28)

    # Create TensorRT engine
    engine = create_tensorrt_engine(model, input_shape, precision="fp16")

    if engine is None:
        print("Failed to create TensorRT engine")
        return

    # Create sample input data
    input_data = np.random.rand(*input_shape).astype(np.float32)

    # Run inference
    output = run_inference_with_engine(engine, input_data)

    print(f"Inference output shape: {output.shape}")
    print(f"Sample output values: {output[:5]}")  # First 5 values

    # Save engine for later use
    with open("optimized_model.trt", 'wb') as f:
        f.write(engine.serialize())

    print("Engine saved to optimized_model.trt")

if __name__ == "__main__":
    main()

Multi-Stream Inference

For humanoid robots requiring high-throughput inference, implement multi-stream processing:

# multi_stream_inference.py
import tensorrt as trt
import pycuda.driver as cuda
import pycuda.autoinit
import numpy as np
import threading
import queue
from concurrent.futures import ThreadPoolExecutor
import time

class MultiStreamTensorRTEngine:
    def __init__(self, engine_path, max_streams=4):
        self.engine = self.load_engine(engine_path)
        self.max_streams = max_streams

        # Create multiple execution contexts and streams
        self.contexts = []
        self.streams = []
        self.bindings = []

        for i in range(max_streams):
            context = self.engine.create_execution_context()
            stream = cuda.Stream()

            self.contexts.append(context)
            self.streams.append(stream)

            # Allocate buffers for each stream
            inputs, outputs, bindings, _ = self.allocate_buffers_for_stream(self.engine)
            self.bindings.append(bindings)

        self.available_streams = queue.Queue()
        for i in range(max_streams):
            self.available_streams.put(i)

        # Performance metrics
        self.inference_times = []
        self.throughput = 0

        print(f"Initialized multi-stream engine with {max_streams} streams")

    def load_engine(self, engine_path):
        """Load TensorRT engine from file"""
        with open(engine_path, 'rb') as f:
            engine_data = f.read()

        runtime = trt.Runtime(trt.Logger(trt.Logger.WARNING))
        engine = runtime.deserialize_cuda_engine(engine_data)

        return engine

    def allocate_buffers_for_stream(self, engine):
        """Allocate buffers for a specific stream"""
        inputs = []
        outputs = []
        bindings = []
        stream = cuda.Stream()  # Temp stream for allocation

        for idx in range(engine.num_bindings):
            binding_shape = engine.get_binding_shape(idx)
            size = trt.volume(binding_shape)
            dtype = trt.nptype(engine.get_binding_dtype(idx))

            host_mem = cuda.pagelocked_empty(size, dtype)
            device_mem = cuda.mem_alloc(host_mem.nbytes)

            bindings.append(int(device_mem))

            if engine.binding_is_input(idx):
                inputs.append({'host': host_mem, 'device': device_mem})
            else:
                outputs.append({'host': host_mem, 'device': device_mem})

        return inputs, outputs, bindings, stream

    def async_inference(self, input_data, stream_id):
        """Perform asynchronous inference on specified stream"""
        start_time = time.time()

        context = self.contexts[stream_id]
        stream = self.streams[stream_id]
        bindings = self.bindings[stream_id]

        # Get input/output bindings
        input_idx = 0  # Assuming first binding is input
        output_idx = 1  # Assuming second binding is output

        # Copy input to GPU
        np.copyto(
            np.ctypeslib.as_array(self.contexts[stream_id]._get_binding_host(0),
                                 shape=(input_data.size,)).reshape(input_data.shape),
            input_data.ravel()
        )

        # Actually, we need to access the proper buffer
        # This is a simplified version - in practice you'd use the allocated buffers
        cuda.memcpy_htod_async(self.bindings[stream_id][0], input_data.ravel(), stream)

        # Execute inference
        context.execute_async_v2(bindings=bindings, stream_handle=stream.handle)

        # Copy output back
        output = np.empty(trt.volume(self.engine.get_binding_shape(1)),
                         dtype=trt.nptype(self.engine.get_binding_dtype(1)))
        cuda.memcpy_dtoh_async(output, self.bindings[stream_id][1], stream)

        # Synchronize stream
        stream.synchronize()

        inference_time = time.time() - start_time
        self.inference_times.append(inference_time)

        return output, inference_time

    def run_parallel_inference(self, input_batch, num_parallel=4):
        """Run parallel inference across multiple streams"""
        futures = []

        with ThreadPoolExecutor(max_workers=num_parallel) as executor:
            for i in range(min(len(input_batch), num_parallel)):
                stream_id = self.available_streams.get()

                future = executor.submit(
                    self.async_inference,
                    input_batch[i],
                    stream_id
                )

                futures.append((future, stream_id))

        # Collect results
        results = []
        for future, stream_id in futures:
            output, time_taken = future.result()
            results.append((output, time_taken))

            # Return stream to available queue
            self.available_streams.put(stream_id)

        return results

    def get_performance_metrics(self):
        """Get current performance metrics"""
        if not self.inference_times:
            return {"avg_time": 0.0, "throughput": 0.0, "latency": 0.0}

        avg_time = sum(self.inference_times[-100:]) / len(self.inference_times[-100:])
        throughput = len(self.inference_times[-100:]) / sum(self.inference_times[-100:])
        latency = avg_time

        return {
            "avg_time": avg_time,
            "throughput": throughput,
            "latency": latency,
            "total_inferences": len(self.inference_times)
        }

        """
        Optimize Grounding DINO model for TensorRT

        Args:
            model_path: Path to ONNX model
            input_shapes: List of tuples (name, min_shape, opt_shape, max_shape)
        """

        network = builder.create_network(1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
        config = builder.create_builder_config()

            if not parser.parse(model_file.read()):
                print("ERROR: Failed to parse the ONNX file.")
                for error in range(parser.num_errors):
                    print(parser.get_error(error))
                return None

        config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE, 4 << 30)  # 4GB

        # Create optimization profile
        profile = builder.create_optimization_profile()

            config.set_flag(trt.BuilderFlag.FP16)

import tensorrt as trt

class Int8Calibrator(trt.IInt8MinMaxCalibrator):
    def __init__(self, calibration_dataset, batch_size=1, algorithm=trt.CalibrationAlgoType.MINMAX_CALIBRATION):
        trt.IInt8MinMaxCalibrator.__init__(self)

        self.dataset = calibration_dataset
        self.batch_size = batch_size
        self.current_index = 0

        # Allocate device memory for calibration batches
        self.device_input = cuda.mem_alloc(self.dataset[0].nbytes)

        self.algorithm = algorithm

    def get_batch_size(self):
        return self.batch_size

    def get_batch(self, names):
        if self.current_index + self.batch_size > len(self.dataset):
            return None

        batch = self.dataset[self.current_index:self.current_index + self.batch_size]
        batch = np.ascontiguousarray(batch)

        cuda.memcpy_htod(self.device_input, batch)

        self.current_index += self.batch_size

        return [int(self.device_input)]

    def read_calibration_cache(self):
        # Try to read calibration cache if it exists
        try:
            with open("calibration.cache", "rb") as f:
                return f.read()
        except FileNotFoundError:
            print("Calibration cache not found, will calibrate from scratch")
            return None

    def write_calibration_cache(self, cache):
        # Write calibration cache to file
        with open("calibration.cache", "wb") as f:
            f.write(cache)

import tensorrt as trt

def configure_for_dla(builder, network, dla_core=0):
    """
    Configure TensorRT engine to use DLA (Deep Learning Accelerator) on Jetson

    Args:
        builder: TensorRT builder
        network: TensorRT network
        dla_core: DLA core to use (0 or 1 on Jetson AGX Orin)
    """
    # Check if DLA is available
    if builder.get_device_type() == trt.DeviceType.DLA:
        print(f"DLA {dla_core} is available")
    else:
        print("DLA not available on this device")
        return False

    # Set DLA usage
    if builder.platform_has_dla:
        # Mark layers that should run on DLA
        # This is typically done by examining the network and marking compatible layers
        for layer in network:
            if is_layer_compatible_with_dla(layer):
                layer.precision = trt.DataType.INT8  # DLA works best with INT8
                layer.device_type = trt.DeviceType.DLA
                layer.epilogue = trt.Epilogue.NONE

    return True

def is_layer_compatible_with_dla(layer):
    """
    Check if a layer is compatible with DLA
    DLA supports a subset of operations
    """
    # DLA supports:
    # - Convolution (with certain constraints)
    # - Fully Connected
    # - Activation functions (ReLU, Sigmoid, Tanh)
    # - Pooling operations
    # - Element-wise operations

    supported_layers = [
        trt.LayerType.CONVOLUTION,
        trt.LayerType.FULLY_CONNECTED,
        trt.LayerType.ACTIVATION,
        trt.LayerType.POOLING,
        trt.LayerType.ELEMENTWISE
    ]

    return layer.type in supported_layers

# tensorrt_profiling.py
import tensorrt as trt
import pycuda.driver as cuda
import pycuda.autoinit
import numpy as np
import time
from collections import defaultdict

class TensorRTProfiler:
    def __init__(self):
        self.profiler = trt.Profiler()
        self.layer_times = defaultdict(list)
        self.total_times = []

    def profile_inference(self, engine, input_data, num_runs=100):
        """Profile inference performance"""
        # Create context with profiler
        context = engine.create_execution_context()
        context.profiler = self.profiler

        # Allocate buffers
        inputs, outputs, bindings, stream = self.allocate_buffers(engine, input_data.shape)

        # Warm up
        for _ in range(10):
            self.run_single_inference(context, input_data, bindings, stream)

        print(f"Profiling {num_runs} inference runs...")

        # Profile runs
        for run in range(num_runs):
            start_time = time.time()

            output = self.run_single_inference(context, input_data, bindings, stream)

            end_time = time.time()
            total_time = end_time - start_time
            self.total_times.append(total_time)

            if run % 20 == 0:  # Print progress
                print(f"Run {run}/{num_runs} completed")

        print(f"Profiling completed for {num_runs} runs")
        return self.get_performance_summary()

    def run_single_inference(self, context, input_data, bindings, stream):
        """Run a single inference with profiling"""
        # Copy input
        np.copyto(inputs[0]['host'], input_data.ravel())
        cuda.memcpy_htod_async(inputs[0]['device'], inputs[0]['host'], stream)

        # Execute
        context.execute_async_v2(bindings=bindings, stream_handle=stream.handle)

        # Copy output
        cuda.memcpy_dtoh_async(outputs[0]['host'], outputs[0]['device'], stream)
        stream.synchronize()

        return outputs[0]['host']

    def get_performance_summary(self):
        """Get performance summary from profiling"""
        if not self.total_times:
            return None

        avg_total_time = sum(self.total_times) / len(self.total_times)

            'total_runs': len(self.total_times),
            'min_time_ms': min(self.total_times) * 1000,
            'max_time_ms': max(self.total_times) * 1000,
            'percentile_95_ms': np.percentile(self.total_times, 95) * 1000,
            'percentile_99_ms': np.percentile(self.total_times, 99) * 1000
        }

        return summary
