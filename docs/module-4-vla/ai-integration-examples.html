AI Pipeline Integration Examples with Whisper, Llama-3, Grounding DINO, and SAM

Introduction

This chapter provides comprehensive code examples that demonstrate the integration of state-of-the-art AI models for humanoid robot applications. We'll explore practical implementations of Whisper for speech recognition, Llama-3 for language understanding, Grounding DINO for open-vocabulary detection, and SAM for precise segmentation. These examples show how to combine these powerful models into a cohesive AI pipeline for humanoid robots.

The integration of these models creates a powerful foundation for vision-language-action (VLA) systems that can understand natural language commands, perceive complex environments, and execute appropriate robotic actions. Each model brings unique capabilities that, when combined, enable sophisticated human-robot interaction.

Complete Integration Pipeline

Main Integration Node

Here's a complete example that integrates all the AI models into a unified pipeline:

#!/usr/bin/env python3
"""
Complete AI Integration Pipeline for Humanoid Robots
Integrates Whisper, Llama-3, Grounding DINO, SAM, and other models
"""
import rospy
import torch
import numpy as np
import cv2
from sensor_msgs.msg import Image, AudioData
from std_msgs.msg import String
from geometry_msgs.msg import PointStamped
from cv_bridge import CvBridge
import openai
import json
import threading
import queue
from typing import Dict, List, Tuple, Optional

# Import AI model libraries
try:
    import whisper
    from transformers import AutoModelForCausalLM, AutoTokenizer
    from segment_anything import sam_model_registry, SamPredictor
    from transformers import AutoProcessor
except ImportError as e:
    rospy.logerr(f"Missing required AI libraries: {e}")
    rospy.logerr("Please install required packages: pip install openai-whisper transformers segment-anything")

class CompleteAIIntegrationPipeline:
    def __init__(self):
        # Initialize ROS node
        rospy.init_node('complete_ai_integration_pipeline')

        # Initialize bridge
        self.bridge = CvBridge()

        # Initialize AI models
        self.initialize_models()

        # Publishers and subscribers
        self.setup_ros_interfaces()

        # Processing queues
        self.audio_queue = queue.Queue(maxsize=10)
        self.image_queue = queue.Queue(maxsize=5)
        self.command_queue = queue.Queue(maxsize=5)

        # Processing threads
        self.processing_thread = None
        self.is_running = True

        # System state
        self.current_environment = {}
        self.conversation_history = []

        rospy.loginfo("Complete AI Integration Pipeline initialized")

    def initialize_models(self):
        """Initialize all AI models"""
        rospy.loginfo("Initializing AI models...")

        # Device configuration
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        rospy.loginfo(f"Using device: {self.device}")

        # 1. Initialize Whisper for speech recognition
        try:
            rospy.loginfo("Loading Whisper model...")
            self.whisper_model = whisper.load_model("base").to(self.device)
        except Exception as e:
            rospy.logerr(f"Failed to load Whisper model: {e}")
            self.whisper_model = None

        # 2. Initialize Llama-3 for language understanding (using a smaller model for edge deployment)
        try:
            rospy.loginfo("Loading language model...")
            # For edge deployment, use a quantized or smaller model
            # In production, you might use a local quantized model
            self.llm_tokenizer = AutoTokenizer.from_pretrained("microsoft/DialoGPT-medium")
            self.llm_model = AutoModelForCausalLM.from_pretrained("microsoft/DialoGPT-medium")
            self.llm_model.to(self.device)
        except Exception as e:
            rospy.logerr(f"Failed to load language model: {e}")
            self.llm_model = None
            self.llm_tokenizer = None

        # 3. Initialize Grounding DINO for open-vocabulary detection
        try:
            rospy.loginfo("Loading Grounding DINO model...")
            # For now, using a placeholder - in practice you'd load the actual model
            self.grounding_dino_model = None  # Placeholder for Grounding DINO
        except Exception as e:
            rospy.logerr(f"Failed to load Grounding DINO model: {e}")
            self.grounding_dino_model = None

        # 4. Initialize SAM for segmentation
        try:
            rospy.loginfo("Loading SAM model...")
            sam_checkpoint = "sam_vit_h_4b8939.pth"  # Download this separately
            self.sam_model = sam_model_registry["vit_h"](checkpoint=sam_checkpoint)
            self.sam_model.to(device=self.device)
            self.sam_predictor = SamPredictor(self.sam_model)
        except Exception as e:
            rospy.logerr(f"Failed to load SAM model: {e}")
            self.sam_model = None
            self.sam_predictor = None

        rospy.loginfo("All models initialized successfully")

    def setup_ros_interfaces(self):
        """Setup ROS publishers and subscribers"""
        # Subscribers
        self.image_sub = rospy.Subscriber("/camera/image_raw", Image, self.image_callback)
        self.audio_sub = rospy.Subscriber("/audio/audio", AudioData, self.audio_callback)
        self.command_sub = rospy.Subscriber("/user_command", String, self.command_callback)

        # Publishers
        self.detection_pub = rospy.Publisher("/object_detections", String, queue_size=10)
        self.response_pub = rospy.Publisher("/robot_response", String, queue_size=10)
        self.action_pub = rospy.Publisher("/robot_action_plan", String, queue_size=10)
        self.status_pub = rospy.Publisher("/ai_pipeline_status", String, queue_size=10)

        rospy.loginfo("ROS interfaces configured")

    def image_callback(self, msg):
        """Handle incoming image messages"""
        try:
            cv_image = self.bridge.imgmsg_to_cv2(msg, "bgr8")

            image_data = {
                'image': cv_image,
                'timestamp': msg.header.stamp.to_sec(),
                'header': msg.header
            }

            if not self.image_queue.full():
                self.image_queue.put(image_data)
            else:
                rospy.logwarn("Image queue full, dropping frame")

        except Exception as e:
            rospy.logerr(f"Error processing image: {e}")

    def audio_callback(self, msg):
        """Handle incoming audio messages"""
        try:
            # Convert audio data to numpy array
            audio_array = np.frombuffer(msg.data, dtype=np.int16).astype(np.float32) / 32768.0

            audio_data = {
                'audio': audio_array,
                'timestamp': msg.header.stamp.to_sec(),
                'header': msg.header
            }

            if not self.audio_queue.full():
                self.audio_queue.put(audio_data)
            else:
                rospy.logwarn("Audio queue full, dropping audio")

        except Exception as e:
            rospy.logerr(f"Error processing audio: {e}")

    def command_callback(self, msg):
        """Handle direct text commands"""
        command_data = {
            'command': msg.data,
            'timestamp': rospy.Time.now().to_sec(),
            'source': 'direct_text'
        }

        if not self.command_queue.full():
            self.command_queue.put(command_data)
        else:
            rospy.logwarn("Command queue full, dropping command")

    def start_processing(self):
        """Start the main processing loop"""
        self.processing_thread = threading.Thread(target=self.processing_loop)
        self.processing_thread.daemon = True
        self.processing_thread.start()

        rospy.loginfo("Processing loop started")

    def processing_loop(self):
        """Main processing loop"""
        rate = rospy.Rate(10)  # 10 Hz processing

        while self.is_running and not rospy.is_shutdown():
            try:
                # Process audio if available
                if not self.audio_queue.empty():
                    audio_data = self.audio_queue.get_nowait()
                    self.process_audio_input(audio_data)

                # Process images if available
                if not self.image_queue.empty():
                    image_data = self.image_queue.get_nowait()
                    self.process_visual_input(image_data)

                # Process commands if available
                if not self.command_queue.empty():
                    command_data = self.command_queue.get_nowait()
                    self.process_command_input(command_data)

                # Publish status
                self.publish_status()

            except queue.Empty:
                pass  # Expected when queues are empty
            except Exception as e:
                rospy.logerr(f"Processing loop error: {e}")

            rate.sleep()

    def process_audio_input(self, audio_data):
        """Process audio input through Whisper"""
        if not self.whisper_model:
            rospy.logwarn("Whisper model not available")
            return

    def process_visual_input(self, image_data):

        try:

    def perform_segmentation(self, image, detections):
        """Perform segmentation using SAM on detected objects"""
        if not self.sam_predictor:
            return detections

        try:
            # Set image for SAM predictor
            self.sam_predictor.set_image(image)

            # Extract bounding boxes for SAM
            boxes = []
            for detection in detections:
                boxes.append(detection['bbox'])

            if not boxes:
                return detections

            # Convert boxes to tensor format
            boxes_tensor = torch.tensor(boxes).float().to(self.device)

            # Transform boxes for SAM
            transformed_boxes = self.sam_predictor.transform.apply_boxes_torch(
                boxes_tensor, image.shape[:2]
            )

            # Predict masks
            masks, scores, logits = self.sam_predictor.predict_torch(
                point_coords=None,
                point_labels=None,
                boxes=transformed_boxes,
                multimask_output=False,
            )

            # Update detections with segmentation masks
            updated_detections = []
            for i, detection in enumerate(detections):
                detection_with_mask = detection.copy()
                if i < len(masks):
                    # Convert mask to numpy array for ROS serialization
                    mask_np = masks[i][0].cpu().numpy()  # Get first mask from batch
                    detection_with_mask['mask'] = mask_np.tolist()  # Convert to list for JSON serialization
                    detection_with_mask['mask_confidence'] = float(scores[i].item())

                updated_detections.append(detection_with_mask)

            return updated_detections

        except Exception as e:
            rospy.logerr(f"SAM segmentation error: {e}")
            return detections

    def process_natural_language_command(self, command, header=None):
        """Process natural language command using LLM"""
        if not self.llm_model or not self.llm_tokenizer:
            rospy.logwarn("Language model not available")
            return

        try:
            # Add to conversation history
            self.conversation_history.append({'role': 'user', 'content': command})

            # Prepare context for the language model
            context = self.build_context_from_environment()

            # Generate response using the language model
            response = self.generate_language_response(command, context)

            if response:
                # Parse the response for robot action
                action_plan = self.parse_response_for_action(response)

                if action_plan:
                    # Publish action plan
                    action_msg = String()
                    action_msg.data = json.dumps({
                        'action_plan': action_plan,
                        'command': command,
                        'response': response,
                        'timestamp': rospy.Time.now().to_sec()
                    })
                    self.action_pub.publish(action_msg)

                # Publish response to user
                response_msg = String()
                response_msg.data = response
                self.response_pub.publish(response_msg)

                # Add to conversation history
                self.conversation_history.append({'role': 'assistant', 'content': response})

                rospy.loginfo(f"Processed command: {command} -> {response}")

        except Exception as e:
            rospy.logerr(f"Language processing error: {e}")

    def build_context_from_environment(self):
        """Build context from current environment"""
        context = {
            'objects': self.current_environment.get('objects', []),
            'location': self.current_environment.get('location', 'unknown'),
            'time': rospy.Time.now().to_sec(),
            'conversation_history': self.conversation_history[-5:]  # Last 5 exchanges
        }
        return context

    def generate_language_response(self, command, context):
        """Generate language response using the LLM"""
        try:
            # For this example, we'll use a simple rule-based approach
            # In practice, you'd use the loaded LLM model

            # Simple keyword-based response generation
            command_lower = command.lower()

            if any(word in command_lower for word in ['hello', 'hi', 'hey']):
                return "Hello! How can I help you today?"
            elif any(word in command_lower for word in ['find', 'where', 'locate']):
                return self.generate_find_response(command, context)
            elif any(word in command_lower for word in ['pick', 'grasp', 'take']):
                return self.generate_grasp_response(command, context)
            elif any(word in command_lower for word in ['go', 'move', 'walk']):
                return self.generate_navigation_response(command, context)
            else:
                return f"I understand you said: '{command}'. How can I assist you with this?"

        except Exception as e:
            rospy.logerr(f"Response generation error: {e}")
            return "I'm sorry, I didn't understand that command."

    def generate_find_response(self, command, context):
        """Generate response for find/locate commands"""
        # Extract object to find from command
        words = command.lower().split()
        object_to_find = None

        for i, word in enumerate(words):
            if word in ['find', 'locate', 'where', 'is']:
                if i + 1 < len(words):
                    object_to_find = words[i + 1]
                    break

        if object_to_find and context.get('objects'):
            # Check if object is detected
            for obj in context['objects']:
                if object_to_find.lower() in obj['label'].lower():
                    bbox = obj['bbox']
                    return f"I found the {obj['label']} at position ({bbox[0]}, {bbox[1]})."

        return f"I couldn't find the {object_to_find or 'object'} in my current view. Should I look around?"

    def generate_grasp_response(self, command, context):
        """Generate response for grasp/pick commands"""
        return "I can help you with that. I'll navigate to the object and grasp it carefully."

    def generate_navigation_response(self, command, context):
        """Generate response for navigation commands"""
        return "I'll navigate to that location. Please wait while I plan my path."

    def parse_response_for_action(self, response):
        """Parse response to generate robot action plan"""
        # Simple action parsing based on response content
        action_plan = []

        if "navigate" in response.lower() or "go to" in response.lower():
            action_plan.append({
                'action': 'navigation',
                'target': 'location_from_command',
                'description': 'Navigate to specified location'
            })

        if "grasp" in response.lower() or "pick" in response.lower():
            action_plan.append({
                'action': 'manipulation',
                'target': 'object_from_perception',
                'description': 'Grasp specified object'
            })

        if not action_plan:
            action_plan.append({
                'action': 'response',
                'content': response,
                'description': 'Respond to user'
            })

        return action_plan

    def process_command_input(self, command_data):
        """Process direct text command input"""
        command = command_data['command']
        rospy.loginfo(f"Processing direct command: {command}")

        # Process as natural language command
        fake_header = None  # No image header for text command
        self.process_natural_language_command(command, fake_header)

    def publish_status(self):
        """Publish system status"""
        status = {
            'timestamp': rospy.Time.now().to_sec(),
            'models_loaded': {
                'whisper': self.whisper_model is not None,
                'llm': self.llm_model is not None,
                'sam': self.sam_model is not None,
                'grounding_dino': self.grounding_dino_model is not None
            },
            'queues': {
                'audio': self.audio_queue.qsize(),
                'image': self.image_queue.qsize(),
                'command': self.command_queue.qsize()
            },
            'environment_objects': len(self.current_environment.get('objects', [])),
            'conversation_length': len(self.conversation_history)
        }

        status_msg = String()
        status_msg.data = json.dumps(status)
        self.status_pub.publish(status_msg)

    def run(self):
        """Main run method"""
        self.start_processing()

        try:
            rospy.spin()
        except rospy.ROSInterruptException:
            rospy.loginfo("AI Integration Pipeline interrupted")
        except KeyboardInterrupt:
            rospy.loginfo("AI Integration Pipeline stopped by user")

        self.is_running = False

Advanced Whisper Processing

# advanced_whisper_integration.py
import whisper
import torch
import numpy as np
import rospy
from audio_common_msgs.msg import AudioData
from std_msgs.msg import String
import threading
import queue

class AdvancedWhisperProcessor:
    def __init__(self, model_size="base"):
        # Initialize Whisper model
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.model = whisper.load_model(model_size).to(self.device)

        # Audio processing parameters
        self.sample_rate = 16000
        self.chunk_duration = 2.0  # Process 2-second chunks
        self.chunk_size = int(self.chunk_duration * self.sample_rate)

        # Audio buffer
        self.audio_buffer = np.array([], dtype=np.float32)
        self.min_audio_duration = 0.5  # Minimum audio duration to process

        # Processing queue
        self.process_queue = queue.Queue(maxsize=5)
        self.result_queue = queue.Queue(maxsize=10)

        # Threading
        self.processing_thread = None
        self.is_processing = True

        # Publishers and subscribers
        self.audio_sub = rospy.Subscriber("/audio/audio", AudioData, self.audio_callback)
        self.transcript_pub = rospy.Publisher("/whisper/transcript", String, queue_size=10)
        self.status_pub = rospy.Publisher("/whisper/status", String, queue_size=10)

        rospy.loginfo("Advanced Whisper Processor initialized")

    def audio_callback(self, msg):
        """Handle incoming audio data"""
        try:
            # Convert audio data to numpy array
            audio_array = np.frombuffer(msg.data, dtype=np.int16).astype(np.float32) / 32768.0

            # Add to buffer
            self.audio_buffer = np.concatenate([self.audio_buffer, audio_array])

            # Process if buffer is large enough
            if len(self.audio_buffer) >= self.chunk_size:
                # Take a chunk and process it
                chunk = self.audio_buffer[:self.chunk_size]
                self.audio_buffer = self.audio_buffer[self.chunk_size:]  # Remove processed chunk

                # Add to processing queue
                if not self.process_queue.full():
                    self.process_queue.put(chunk)
                else:
                    rospy.logwarn("Processing queue full, dropping audio chunk")

        except Exception as e:
            rospy.logerr(f"Error in audio callback: {e}")

    def start_processing(self):
        """Start the processing thread"""
        self.processing_thread = threading.Thread(target=self.processing_loop)
        self.processing_thread.daemon = True
        self.processing_thread.start()

        rospy.loginfo("Whisper processing thread started")

    def processing_loop(self):
        """Main processing loop for Whisper transcription"""
        while self.is_processing:
            try:
                # Get audio chunk from queue
                audio_chunk = self.process_queue.get(timeout=0.1)

                # Transcribe the chunk
                transcript = self.transcribe_chunk(audio_chunk)

                if transcript and len(transcript.strip()) > 3:
                    # Publish transcript
                    transcript_msg = String()
                    transcript_msg.data = transcript.strip()
                    self.transcript_pub.publish(transcript_msg)

                    rospy.loginfo(f"Transcribed: {transcript.strip()}")

            except queue.Empty:
                continue
            except Exception as e:
                rospy.logerr(f"Whisper processing error: {e}")

    def transcribe_chunk(self, audio_chunk):
        """Transcribe a single audio chunk"""
        try:
            # Ensure minimum audio duration
            if len(audio_chunk) < self.min_audio_duration * self.sample_rate:
                return None

            # Transcribe using Whisper
            result = self.model.transcribe(
                audio_chunk,
                language="en",
                temperature=0.0,  # Deterministic output
                compression_ratio_threshold=2.4,
                logprob_threshold=-1.0,
                no_speech_threshold=0.6
            )

            return result["text"]

        except Exception as e:
            rospy.logerr(f"Transcription error: {e}")
            return None

    def transcribe_with_options(self, audio_data, options=None):
        """Transcribe with specific options"""
        if options is None:
            options = {}

        try:
            # Apply options
            temperature = options.get('temperature', 0.0)
            language = options.get('language', 'en')
            prompt = options.get('prompt', '')

            result = self.model.transcribe(
                audio_data,
                language=language,
                temperature=temperature,
                initial_prompt=prompt
            )

            return result["text"]

        except Exception as e:
            rospy.logerr(f"Optioned transcription error: {e}")
            return None

    def get_status(self):
        """Get current processing status"""
        status = {
            'buffer_size': len(self.audio_buffer),
            'queue_sizes': {
                'process': self.process_queue.qsize(),
                'result': self.result_queue.qsize()
            },
            'model_loaded': self.model is not None,
            'device': str(self.device)
        }
        return status

    def run(self):
        """Run the processor"""
        self.start_processing()

        rate = rospy.Rate(1)  # 1 Hz status publishing

        while not rospy.is_shutdown():
            # Publish status
            status_msg = String()
            status_msg.data = str(self.get_status())
            self.status_pub.publish(status_msg)

            rate.sleep()

def whisper_main():
    rospy.init_node('advanced_whisper_processor')
    processor = AdvancedWhisperProcessor()
    processor.run()

if __name__ == "__main__":
    whisper_main()

        """Segment detections using SAM"""
        if not self.sam_predictor or not detections:
            return detections

        try:
            # Set image for SAM predictor
            self.sam_predictor.set_image(image)

            # Extract bounding boxes
            boxes = []
            for detection in detections:
                boxes.append(detection['bbox'])

            if not boxes:
                return detections

            # Convert boxes to tensor
            boxes_tensor = torch.tensor(boxes).float().to(self.device)

            # Transform boxes
            transformed_boxes = self.sam_predictor.transform.apply_boxes_torch(
                boxes_tensor, image.shape[:2]
            )

            # Predict masks
            masks, scores, logits = self.sam_predictor.predict_torch(
                point_coords=None,
                point_labels=None,
                boxes=transformed_boxes,
                multimask_output=False,
            )

            # Update detections with masks
            updated_detections = []
            for i, detection in enumerate(detections):
                detection_with_mask = detection.copy()
                if i < len(masks):
                    mask_np = masks[i][0].cpu().numpy()
                    detection_with_mask['mask'] = mask_np.tolist()
                    detection_with_mask['mask_confidence'] = float(scores[i].item())

                updated_detections.append(detection_with_mask)

            return updated_detections

        except Exception as e:
            rospy.logerr(f"SAM segmentation error: {e}")
            return detections

            cv2.rectangle(vis_image, (x1, y1), (x2, y2), (0, 255, 0), 2)

            if mask:

                mask_colored = np.zeros_like(vis_image)
                mask_colored[mask_np > 0.5] = [255, 0, 0]  # Red mask

import torch

import rospy

class Llama3Integration:
    def __init__(self, model_name="microsoft/DialoGPT-medium"):
        # Initialize ROS
        rospy.init_node('llama3_integration')

        # Initialize model (using a smaller model for edge deployment)
        self.model_name = model_name
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        # Use a lightweight model for edge deployment
        # In practice, you might use a quantized version
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.model.to(self.device)

        # Add padding token if not present
        if self.tokenizer.pad_token is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token

        # Conversation history
        self.chat_history_ids = None
        self.max_history_length = 1000  # Limit conversation history

        # Setup ROS interfaces
        self.setup_ros_interfaces()

        # Processing queue
        self.command_queue = queue.Queue(maxsize=10)
        self.response_queue = queue.Queue(maxsize=10)

        rospy.loginfo(f"Llama-3 Integration initialized on {self.device}")

    def setup_ros_interfaces(self):
        """Setup ROS publishers and subscribers"""
        # Subscribers
        self.command_sub = rospy.Subscriber("/llm_input", String, self.command_callback)

        # Publishers
        self.response_pub = rospy.Publisher("/llm_response", String, queue_size=10)
        self.status_pub = rospy.Publisher("/llm_status", String, queue_size=10)

    def command_callback(self, msg):
        """Handle incoming commands"""
        try:
            command_data = json.loads(msg.data) if msg.data.startswith('{') else {'command': msg.data}
            command = command_data.get('command', msg.data)

            if not self.command_queue.full():
                self.command_queue.put({
                    'command': command,
                    'timestamp': rospy.Time.now().to_sec(),
                    'original_msg': msg
                })
            else:
                rospy.logwarn("Command queue full, dropping message")

        except json.JSONDecodeError:
            # If not JSON, treat as plain command
            if not self.command_queue.full():
                self.command_queue.put({
                    'command': msg.data,
                    'timestamp': rospy.Time.now().to_sec(),
                    'original_msg': msg
                })

    def process_commands(self):
        """Process commands in a separate thread"""
        while not rospy.is_shutdown():
            try:
                # Get command from queue
                command_data = self.command_queue.get(timeout=0.1)

                # Generate response
                response = self.generate_response(command_data['command'])

                if response:
                    # Publish response
                    response_msg = String()
                    response_msg.data = json.dumps({
                        'response': response,
                        'input_command': command_data['command'],
                        'timestamp': rospy.Time.now().to_sec()
                    })
                    self.response_pub.publish(response_msg)

                    rospy.loginfo(f"Processed: {command_data['command']} -> {response[:50]}...")

            except queue.Empty:
                continue
            except Exception as e:
                rospy.logerr(f"Command processing error: {e}")

    def generate_response(self, user_input):
        """Generate response using the LLM"""
        try:
            # Encode user input
            new_user_input_ids = self.tokenizer.encode(
                user_input + self.tokenizer.eos_token,
                return_tensors='pt'
            ).to(self.device)

            # Append to chat history
            if self.chat_history_ids is not None:
                bot_input_ids = torch.cat([self.chat_history_ids, new_user_input_ids], dim=-1)
            else:
                bot_input_ids = new_user_input_ids

            # Generate response
            self.chat_history_ids = self.model.generate(
                bot_input_ids,
                max_length=1000,
                num_beams=5,
                no_repeat_ngram_size=3,
                early_stopping=True,
                pad_token_id=self.tokenizer.eos_token_id,
                temperature=0.7,
                do_sample=True
            )

            # Decode response
            response = self.tokenizer.decode(
                self.chat_history_ids[:, bot_input_ids.shape[-1]:][0],
                skip_special_tokens=True
            )

            # Limit conversation history to prevent memory issues
            if self.chat_history_ids.shape[-1] > self.max_history_length:
                self.chat_history_ids = self.chat_history_ids[:, -self.max_history_length:]

            return response

        except Exception as e:
            rospy.logerr(f"Response generation error: {e}")
            return "I'm sorry, I encountered an error processing your request."

    def reset_conversation(self):
        """Reset conversation history"""
        self.chat_history_ids = None

    def run(self):
        """Run the LLM integration"""
        # Start processing thread
        processing_thread = threading.Thread(target=self.process_commands)
        processing_thread.daemon = True
        processing_thread.start()

        # Main loop for status updates
        rate = rospy.Rate(1)  # 1 Hz status updates

        while not rospy.is_shutdown():
            # Publish status
            status = {
                'model_loaded': self.model is not None,
                'device': str(self.device),
                'queue_sizes': {
                    'commands': self.command_queue.qsize(),
                    'responses': self.response_queue.qsize()
                },
                'conversation_length': self.chat_history_ids.shape[-1] if self.chat_history_ids is not None else 0
            }

            status_msg = String()
            status_msg.data = json.dumps(status)
            self.status_pub.publish(status_msg)

            rate.sleep()

def llama3_main():
    try:
        node = Llama3Integration()

    except rospy.ROSInterruptException:

        """Create TensorRT engine from ONNX model"""
        try:
            # Create builder
            builder = trt.Builder(self.logger)
            network = builder.create_network(1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
            config = builder.create_builder_config()

            # Set memory limits appropriate for Jetson
            config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE, 2 << 30)  # 2GB

            # Create parser
            parser = trt.OnnxParser(network, self.logger)

            # Parse ONNX model
            with open(onnx_model_path, 'rb') as model_file:
                if not parser.parse(model_file.read()):
                    rospy.logerr("Failed to parse ONNX model")
                    for error in range(parser.num_errors):
                        rospy.logerr(parser.get_error(error))
                    return False

            # Create optimization profile
            profile = builder.create_optimization_profile()

            # Set input shapes for the profile
            for i, (input_name, min_shape, opt_shape, max_shape) in enumerate(input_shapes):
                profile.set_shape(input_name, min_shape, opt_shape, max_shape)

            config.add_optimization_profile(profile)

            # Build engine
            engine = builder.build_engine(network, config)
            if engine is None:
                rospy.logerr("Failed to build TensorRT engine")
                return False

            # Serialize engine
            with open(engine_path, 'wb') as f:
                f.write(engine.serialize())

            rospy.loginfo(f"TensorRT engine created: {engine_path}")
            return True

Performance Analysis Tools

# performance_analysis.py
import rospy
import time
from std_msgs.msg import String
import json
import psutil
import GPUtil

class PerformanceAnalyzer:
    def __init__(self):
        rospy.init_node('performance_analyzer')

        self.metrics_pub = rospy.Publisher('/performance/metrics', String, queue_size=10)

        self.start_times = {}
        self.metrics_history = {
            'cpu_percent': [],
            'memory_percent': [],
            'gpu_percent': [],
            'process_times': []
        }

        rospy.loginfo("Performance Analyzer initialized")

    def start_timer(self, operation_name):
        """Start timing for an operation"""
        self.start_times[operation_name] = time.time()

    def stop_timer(self, operation_name):
        """Stop timing and record"""
        if operation_name in self.start_times:
            elapsed = time.time() - self.start_times[operation_name]
            self.metrics_history['process_times'].append({
                'operation': operation_name,
                'time': elapsed,
                'timestamp': rospy.Time.now().to_sec()
            })
            return elapsed
        return None

    def get_system_metrics(self):
        """Get current system metrics"""
        metrics = {
            'timestamp': rospy.Time.now().to_sec(),
            'cpu_percent': psutil.cpu_percent(interval=1),
            'memory_percent': psutil.virtual_memory().percent,
            'memory_available': psutil.virtual_memory().available,
            'disk_percent': psutil.disk_usage('/').percent,
        }

        # GPU metrics if available
        gpus = GPUtil.getGPUs()
        if gpus:
            gpu = gpus[0]  # Assuming single GPU system
            metrics.update({
                'gpu_percent': gpu.load * 100,
                'gpu_memory_percent': gpu.memoryUtil * 100,
                'gpu_temperature': gpu.temperature,
                'gpu_memory_used': gpu.memoryUsed,
                'gpu_memory_total': gpu.memoryTotal
            })

        return metrics

    def run(self):
        """Run performance monitoring"""
        rate = rospy.Rate(1)  # 1 Hz monitoring

        while not rospy.is_shutdown():
            # Get system metrics
            metrics = self.get_system_metrics()

            # Add to history
            self.metrics_history['cpu_percent'].append(metrics['cpu_percent'])
            self.metrics_history['memory_percent'].append(metrics['memory_percent'])

            if 'gpu_percent' in metrics:
                self.metrics_history['gpu_percent'].append(metrics['gpu_percent'])

            # Keep only recent history (last 100 measurements)
            for key in self.metrics_history:
                if len(self.metrics_history[key]) > 100:
                    self.metrics_history[key] = self.metrics_history[key][-100:]

            # Publish metrics
            metrics_msg = String()
            metrics_msg.data = json.dumps({
                'current': metrics,
                'history': self.metrics_history,
                'averages': {
                    'cpu_avg': sum(self.metrics_history['cpu_percent'][-10:]) / len(self.metrics_history['cpu_percent'][-10:]) if self.metrics_history['cpu_percent'] else 0,
                    'memory_avg': sum(self.metrics_history['memory_percent'][-10:]) / len(self.metrics_history['memory_percent'][-10:]) if self.metrics_history['memory_percent'] else 0,
                    'gpu_avg': sum(self.metrics_history['gpu_percent'][-10:]) / len(self.metrics_history['gpu_percent'][-10:]) if self.metrics_history['gpu_percent'] else 0,
                }
            })

            self.metrics_pub.publish(metrics_msg)

            rate.sleep()

def performance_main():
    analyzer = PerformanceAnalyzer()
    analyzer.run()

if __name__ == "__main__":
    performance_main()
