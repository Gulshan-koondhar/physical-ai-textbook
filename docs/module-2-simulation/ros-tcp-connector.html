ROS TCP Connector Bridge for Unity Integration

Introduction

The ROS TCP Connector provides a bridge between ROS 2 and Unity, enabling real-time communication for humanoid robotics simulation and visualization. This bridge allows Unity to act as a high-quality rendering environment while ROS 2 handles the physics simulation, control systems, and sensor processing. This chapter covers the setup, configuration, and usage of the ROS TCP Connector for humanoid robotics applications.

Understanding the ROS TCP Connector Architecture

Core Components

The ROS TCP Connector consists of several key components:

  1. ROS Bridge Node: Runs in the ROS 2 environment and handles message conversion
  2. TCP Communication Layer: Provides low-latency communication between ROS 2 and Unity
  3. Unity ROS Package: Handles message reception and Unity scene updates
  4. Message Converters: Translate between ROS message types and Unity data structures

Communication Flow

The typical communication flow is:

  • ROS 2 nodes publish messages to topics
  • ROS Bridge receives messages and sends them over TCP to Unity
  • Unity receives messages and updates the 3D scene
  • Unity can publish messages back to ROS 2 through the bridge

Setting up the ROS TCP Connector

Prerequisites

Before setting up the ROS TCP Connector, ensure you have:

  1. ROS 2 Humble Hawksbill installed and configured
  2. Unity 2021.3 LTS or newer with Unity Hub
  3. Python 3.8+ for the bridge node
  4. Basic knowledge of ROS 2 concepts (topics, messages, services)

Installing the ROS Bridge

Method 1: Using APT (Ubuntu)

sudo apt update
sudo apt install ros-humble-rosbridge-suite

Method 2: Building from Source

# Create a workspace
mkdir -p ~/rosbridge_ws/src
cd ~/rosbridge_ws/src

# Clone the rosbridge repository
git clone -b humble https://github.com/RobotWebTools/rosbridge_suite.git

# Build the workspace
cd ~/rosbridge_ws
colcon build --packages-select rosbridge_suite
source install/setup.bash

Installing Unity ROS TCP Connector

In Unity, install the ROS TCP Connector package:

  1. Open Unity Hub and create a new 3D project
  2. Go to Window → Package Manager
  3. Click the "+" button → Add package from git URL
  4. Add: com.unity.robotics.ros-tcp-connector

Configuring the ROS Bridge

Basic Bridge Launch File

Create a launch file to start the ROS bridge:

# launch/rosbridge.launch.py
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node

def generate_launch_description():
    # Launch configuration variables
    port = LaunchConfiguration('port', default='9090')
    address = LaunchConfiguration('address', default='127.0.0.1')
    reuse_service = LaunchConfiguration('reuse_service', default='true')

    # Launch arguments
    declare_port_cmd = DeclareLaunchArgument(
        'port',
        default_value='9090',
        description='Port to use for the ROS bridge')

    declare_address_cmd = DeclareLaunchArgument(
        'address',
        default_value='127.0.0.1',
        description='Address to use for the ROS bridge')

    declare_reuse_service_cmd = DeclareLaunchArgument(
        'reuse_service',
        default_value='true',
        description='Whether to reuse service connections')

    # ROS Bridge Server node
    rosbridge_server = Node(
        package='rosbridge_server',
        executable='rosbridge_websocket',
        name='rosbridge_server',
        parameters=[
            {'port': port},
            {'address': address},
            {'reuse_service_client': reuse_service},
            {'authenticate_services': False},
            {'fragment_size': 4096},
            {'max_message_size': 65536},
            {'unregister_timeout': 10.0}
        ],
        output='screen'
    )

    # Create launch description
    ld = LaunchDescription()

    # Add launch arguments
    ld.add_action(declare_port_cmd)
    ld.add_action(declare_address_cmd)
    ld.add_action(declare_reuse_service_cmd)

    # Add ROS bridge node
    ld.add_action(rosbridge_server)

    return ld

Unity Integration Setup

Basic Unity Connection Script

Create a connection script in Unity to connect to the ROS bridge:

// Scripts/ROSConnectionManager.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Sensor;
using RosMessageTypes.Geometry;

public class ROSConnectionManager : MonoBehaviour
{
    [Header("ROS Connection Settings")]
    [SerializeField] private string rosIPAddress = "127.0.0.1";
    [SerializeField] private int rosPort = 9090;
    [SerializeField] private bool useSSL = false;

    [Header("Subscribed Topics")]
    [SerializeField] private string jointStatesTopic = "/joint_states";
    [SerializeField] private string imuTopic = "/imu/data";
    [SerializeField] private string odometryTopic = "/odom";

    private ROSConnection rosConnection;
    private bool isConnected = false;

    void Start()
    {
        InitializeROSConnection();
        SubscribeToTopics();
    }

    void InitializeROSConnection()
    {
        rosConnection = ROSConnection.GetOrCreateInstance();
        rosConnection.OnConnected += OnConnected;
        rosConnection.OnDisconnected += OnDisconnected;

        // Configure connection
        rosConnection.host = rosIPAddress;
        rosConnection.port = rosPort;
        rosConnection.useSSL = useSSL;
    }

    void SubscribeToTopics()
    {
        // Subscribe to joint states for humanoid robot visualization
        rosConnection.Subscribe<JointStateMsg>(jointStatesTopic, OnJointStateReceived);

        // Subscribe to IMU data for orientation visualization
        rosConnection.Subscribe<ImuMsg>(imuTopic, OnIMUReceived);

        // Subscribe to odometry for position tracking
        rosConnection.Subscribe<OdometryMsg>(odometryTopic, OnOdometryReceived);
    }

    void OnConnected()
    {
        isConnected = true;
        Debug.Log("Connected to ROS bridge at " + rosConnection.host + ":" + rosConnection.port);
    }

    void OnDisconnected()
    {
        isConnected = false;
        Debug.LogWarning("Disconnected from ROS bridge");
    }

    void OnJointStateReceived(JointStateMsg msg)
    {
        // Process joint state message
        var jointPositions = new System.Collections.Generic.Dictionary<string, double>();

        for (int i = 0; i < msg.name.Count; i++)
        {
            if (i < msg.position.Count)
            {
                jointPositions[msg.name[i]] = msg.position[i];
            }
        }

        // Notify other components about the new joint states
        SendMessage("UpdateRobotJoints", jointPositions, SendMessageOptions.DontRequireReceiver);
    }

    void OnIMUReceived(ImuMsg msg)
    {
        // Process IMU data
        Vector3 orientation = new Vector3(
            (float)msg.orientation.x,
            (float)msg.orientation.y,
            (float)msg.orientation.z
        );

        // Use for visualization or other purposes
        SendMessage("UpdateOrientation", orientation, SendMessageOptions.DontRequireReceiver);
    }

    void OnOdometryReceived(OdometryMsg msg)
    {
        // Process odometry data
        Vector3 position = new Vector3(
            (float)msg.pose.pose.position.x,
            (float)msg.pose.pose.position.z,  // Unity Y is up
            (float)msg.pose.pose.position.y   // Swap Y and Z
        );

        // Update robot position in Unity
        transform.position = position;
    }

    public bool IsConnected()
    {
        return isConnected;
    }

    void OnApplicationQuit()
    {
        if (rosConnection != null)
        {
            rosConnection.Close();
        }
    }
}

Humanoid Robot Joint Controller

Create a script to update the humanoid robot's joints based on ROS messages:

// Scripts/HumanoidJointController.cs
using UnityEngine;
using System.Collections.Generic;

[System.Serializable]
public class JointMapping
{
    public string rosJointName;
    public Transform unityJoint;
    public JointType jointType;
    public Vector3 rotationAxis = Vector3.up;
    public float minAngle = -180f;
    public float maxAngle = 180f;
    public float positionMultiplier = 1f;
}

public enum JointType
{
    Revolute,
    Prismatic,
    Fixed
}

public class HumanoidJointController : MonoBehaviour
{
    [Header("Joint Configuration")]
    [SerializeField] private List<JointMapping> jointMappings = new List<JointMapping>();
    [SerializeField] private bool updateInFixedUpdate = true;

    private Dictionary<string, double> currentJointPositions = new Dictionary<string, double>();
    private bool hasNewData = false;

    void Start()
    {
        // Initialize all joint positions to zero if not already mapped
        foreach (var mapping in jointMappings)
        {
            if (!currentJointPositions.ContainsKey(mapping.rosJointName))
            {
                currentJointPositions[mapping.rosJointName] = 0.0;
            }
        }
    }

    void Update()
    {
        if (!updateInFixedUpdate)
        {
            UpdateJoints();
        }
    }

    void FixedUpdate()
    {
        if (updateInFixedUpdate)
        {
            UpdateJoints();
        }
    }

    public void UpdateRobotJoints(Dictionary<string, double> jointPositions)
    {
        currentJointPositions = new Dictionary<string, double>(jointPositions);
        hasNewData = true;
    }

    void UpdateJoints()
    {
        if (!hasNewData) return;

        foreach (var mapping in jointMappings)
        {
            if (currentJointPositions.ContainsKey(mapping.rosJointName))
            {
                double position = currentJointPositions[mapping.rosJointName];
                ApplyJointTransform(mapping, (float)position);
            }
        }

        hasNewData = false;
    }

    void ApplyJointTransform(JointMapping mapping, float position)
    {
        if (mapping.unityJoint == null) return;

        switch (mapping.jointType)
        {
            case JointType.Revolute:
                // Apply rotational joint
                float clampedPosition = Mathf.Clamp(position, mapping.minAngle * Mathf.Deg2Rad, mapping.maxAngle * Mathf.Deg2Rad);
                float rotationInDegrees = clampedPosition * Mathf.Rad2Deg * mapping.positionMultiplier;

                Vector3 rotation = mapping.rotationAxis * rotationInDegrees;
                mapping.unityJoint.localEulerAngles = rotation;
                break;

            case JointType.Prismatic:
                // Apply linear joint
                float linearPosition = position * mapping.positionMultiplier;
                Vector3 linearOffset = mapping.rotationAxis * linearPosition;
                mapping.unityJoint.localPosition += linearOffset;
                break;

            case JointType.Fixed:
                // No movement for fixed joints
                break;
        }
    }

    // Helper method to add joint mapping at runtime
    public void AddJointMapping(string rosJointName, Transform unityJoint, JointType jointType = JointType.Revolute)
    {
        JointMapping newMapping = new JointMapping
        {
            rosJointName = rosJointName,
            unityJoint = unityJoint,
            jointType = jointType
        };

        jointMappings.Add(newMapping);
        currentJointPositions[rosJointName] = 0.0;
    }
}

Publishing Messages from Unity

Joint Command Publisher

Create a script to publish joint commands from Unity back to ROS:

// Scripts/JointCommandPublisher.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Sensor;
using RosMessageTypes.Std;

public class JointCommandPublisher : MonoBehaviour
{
    [Header("Publisher Settings")]
    [SerializeField] private string jointCommandTopic = "/joint_commands";
    [SerializeField] private float publishRate = 50f; // Hz

    [Header("Joint Configuration")]
    [SerializeField] private string[] jointNames;
    [SerializeField] private Transform[] jointTransforms;

    private ROSConnection ros;
    private float lastPublishTime = 0f;
    private float publishInterval;

    void Start()
    {
        ros = ROSConnection.GetOrCreateInstance();
        publishInterval = 1f / publishRate;

        if (jointNames.Length != jointTransforms.Length)
        {
            Debug.LogError("Joint names and transforms arrays must have the same length!");
        }
    }

    void Update()
    {
        if (Time.time - lastPublishTime >= publishInterval)
        {
            PublishJointCommands();
            lastPublishTime = Time.time;
        }
    }

    void PublishJointCommands()
    {
        if (jointNames.Length == 0 || jointTransforms.Length == 0) return;

        // Create joint state message
        var jointState = new JointStateMsg();
        jointState.header = new HeaderMsg();
        jointState.header.stamp = new TimeStamp(ROSConnection.GetNodeTimeInSecs());
        jointState.header.frame_id = "base_link";

        // Add joint names and positions
        for (int i = 0; i < jointNames.Length && i < jointTransforms.Length; i++)
        {
            jointState.name.Add(jointNames[i]);

            // Convert Unity rotation to ROS joint position (simplified)
            // This depends on your joint configuration
            float jointPosition = jointTransforms[i].localEulerAngles.y * Mathf.Deg2Rad;
            jointState.position.Add(jointPosition);
        }

        // Publish the message
        ros.Publish(jointCommandTopic, jointState);
    }

    // Method to publish specific joint positions
    public void PublishJointPositions(Dictionary<string, float> jointPositions)
    {
        var jointState = new JointStateMsg();
        jointState.header = new HeaderMsg();
        jointState.header.stamp = new TimeStamp(ROSConnection.GetNodeTimeInSecs());
        jointState.header.frame_id = "base_link";

        foreach (var kvp in jointPositions)
        {
            jointState.name.Add(kvp.Key);
            jointState.position.Add(kvp.Value);
        }

        ros.Publish(jointCommandTopic, jointState);
    }
}

Velocity Command Publisher

For humanoid locomotion, create a velocity command publisher:

// Scripts/VelocityCommandPublisher.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Geometry;

public class VelocityCommandPublisher : MonoBehaviour
{
    [Header("Topic Configuration")]
    [SerializeField] private string cmdVelTopic = "/cmd_vel";
    [SerializeField] private float maxLinearVelocity = 1.0f;
    [SerializeField] private float maxAngularVelocity = 1.0f;

    private ROSConnection ros;

    void Start()
    {
        ros = ROSConnection.GetOrCreateInstance();
    }

    public void PublishVelocityCommand(float linearX, float angularZ)
    {
        // Clamp velocities to prevent excessive commands
        linearX = Mathf.Clamp(linearX, -maxLinearVelocity, maxLinearVelocity);
        angularZ = Mathf.Clamp(angularZ, -maxAngularVelocity, maxAngularVelocity);

        var twist = new TwistMsg();
        twist.linear = new Vector3Msg(linearX, 0, 0);
        twist.angular = new Vector3Msg(0, 0, angularZ);

        ros.Publish(cmdVelTopic, twist);
    }

    // Publish velocity based on Unity input
    public void PublishFromInput()
    {
        float linearInput = Input.GetAxis("Vertical");
        float angularInput = Input.GetAxis("Horizontal");

        PublishVelocityCommand(linearInput * maxLinearVelocity, angularInput * maxAngularVelocity);
    }
}

// Scripts/ROSServiceCaller.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Std;

public class ROSServiceCaller : MonoBehaviour
{
    [Header("Service Configuration")]
    [SerializeField] private string serviceName = "/set_pose";
    [SerializeField] private float timeoutSec = 5.0f;

    private ROSConnection ros;

    void Start()
    {
        ros = ROSConnection.GetOrCreateInstance();
    }

    public void CallSetPoseService(string frameId, float x, float y, float z)
    {
        var request = new SetPoseRequest();
        request.target_pose = new RosMessageTypes.Geometry.PoseMsg();
        request.target_pose.position = new RosMessageTypes.Geometry.PointMsg(x, y, z);
        request.target_pose.orientation = new RosMessageTypes.Geometry.QuaternionMsg(0, 0, 0, 1);

        ros.CallService<SetPoseRequest, SetPoseResponse>(
            serviceName,
            request,
            (SetPoseResponse response) =>
            {
                Debug.Log("Service call successful: " + response.success);
            },
            (float)timeoutSec
        );
    }

    public void CallEmptyService(string serviceToCall)
    {
        var request = new EmptyRequest();

        ros.CallService<EmptyRequest, EmptyResponse>(
            serviceToCall,
            request,
            (EmptyResponse response) =>
            {
                Debug.Log("Empty service call completed");
            },
            (float)timeoutSec
        );
    }
}

Performance Optimization

Efficient Message Handling

Optimize for humanoid robots with many joints:

// Scripts/OptimizedJointHandler.cs
using UnityEngine;
using System.Collections.Generic;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Sensor;

public class OptimizedJointHandler : MonoBehaviour
{
    [Header("Optimization Settings")]
    [SerializeField] private float updateInterval = 0.02f; // 50Hz
    [SerializeField] private bool useInterpolation = true;

    private Dictionary<string, double> currentJointPositions = new Dictionary<string, double>();
    private Dictionary<string, double> targetJointPositions = new Dictionary<string, double>();
    private Dictionary<string, double> previousJointPositions = new Dictionary<string, double>();

    private float lastUpdateTime = 0f;
    private HumanoidJointController jointController;

    void Start()
    {
        jointController = GetComponent<HumanoidJointController>();
        rosConnection = ROSConnection.GetOrCreateInstance();
        rosConnection.Subscribe<JointStateMsg>("/joint_states", OnJointStateReceived);
    }

    void OnJointStateReceived(JointStateMsg msg)
    {
        // Store previous positions for interpolation
        previousJointPositions = new Dictionary<string, double>(currentJointPositions);

        // Update target positions
        for (int i = 0; i < msg.name.Count && i < msg.position.Count; i++)
        {
            targetJointPositions[msg.name[i]] = msg.position[i];
        }

        lastUpdateTime = Time.time;
    }

    void Update()
    {
        if (useInterpolation && Time.time - lastUpdateTime < updateInterval)
        {
            float t = (Time.time - lastUpdateTime) / updateInterval;
            t = Mathf.Clamp01(t);

            // Interpolate between previous and target positions
            var interpolatedPositions = new Dictionary<string, double>();

            foreach (var kvp in targetJointPositions)
            {
                double prevPos = previousJointPositions.ContainsKey(kvp.Key) ?
                                previousJointPositions[kvp.Key] : kvp.Value;

                double interpolatedPos = prevPos + (kvp.Value - prevPos) * t;
                interpolatedPositions[kvp.Key] = interpolatedPos;
            }

            jointController.UpdateRobotJoints(interpolatedPositions);
        }
        else if (Time.time - lastUpdateTime >= updateInterval)
        {
            // Use target positions directly if interpolation is disabled or update interval passed
            jointController.UpdateRobotJoints(new Dictionary<string, double>(targetJointPositions));
        }
    }
}

Connection Status Monitoring

Monitor and handle connection issues gracefully:

// Scripts/ConnectionMonitor.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using System.Collections;

public class ConnectionMonitor : MonoBehaviour
{
    [Header("Connection Settings")]
    [SerializeField] private float connectionCheckInterval = 1.0f;
    [SerializeField] private int maxReconnectionAttempts = 5;
    [SerializeField] private float reconnectionDelay = 2.0f;

    private ROSConnection rosConnection;
    private bool isConnected = false;
    private int reconnectionAttempts = 0;
    private Coroutine reconnectionCoroutine;

    void Start()
    {
        rosConnection = ROSConnection.GetOrCreateInstance();
        rosConnection.OnConnected += OnConnected;
        rosConnection.OnDisconnected += OnDisconnected;

        StartCoroutine(CheckConnectionStatus());
    }

    IEnumerator CheckConnectionStatus()
    {
        while (true)
        {
            yield return new WaitForSeconds(connectionCheckInterval);

            if (rosConnection != null)
            {
                // Check if we're still connected
                if (!isConnected && reconnectionAttempts < maxReconnectionAttempts)
                {
                    AttemptReconnection();
                }
            }
        }
    }

    void OnConnected()
    {
        isConnected = true;
        reconnectionAttempts = 0;
        Debug.Log("Successfully connected to ROS bridge");

        if (reconnectionCoroutine != null)
        {
            StopCoroutine(reconnectionCoroutine);
        }
    }

    void OnDisconnected()
    {
        isConnected = false;
        Debug.LogWarning("Disconnected from ROS bridge. Attempting to reconnect...");

        if (reconnectionAttempts < maxReconnectionAttempts)
        {
            reconnectionCoroutine = StartCoroutine(ReconnectionRoutine());
        }
        else
        {
            Debug.LogError("Maximum reconnection attempts reached. Please check ROS bridge connection.");
        }
    }

    void AttemptReconnection()
    {
        if (rosConnection != null)
        {
            rosConnection.Connect();
        }
    }

    IEnumerator ReconnectionRoutine()
    {
        while (reconnectionAttempts < maxReconnectionAttempts && !isConnected)
        {
            reconnectionAttempts++;
            Debug.Log($"Reconnection attempt {reconnectionAttempts}/{maxReconnectionAttempts}");

            rosConnection.Connect();

            yield return new WaitForSeconds(reconnectionDelay);
        }

        if (!isConnected)
        {
            Debug.LogError("Failed to reconnect after maximum attempts");
        }
    }

    public bool IsConnected()
    {
        return isConnected;
    }

    void OnApplicationQuit()
    {
        if (rosConnection != null)
        {
            rosConnection.Close();
        }
    }
}

Troubleshooting Common Issues

Connection Problems

Issue: Cannot Connect to ROS Bridge

Solutions:

  • Verify ROS bridge is running: ros2 run rosbridge_server rosbridge_websocket
  • Check IP address and port settings
  • Ensure firewall allows connections on the specified port
  • Verify network connectivity between Unity and ROS systems

Issue: High Latency

Solutions:

  • Use TCP_NO_DELAY option in bridge configuration
  • Reduce message frequency where possible
  • Optimize network configuration
  • Consider local deployment for minimum latency

Message Handling Issues

Issue: Messages Not Receiving

Solutions:

  • Check topic names match between ROS and Unity
  • Verify message types are compatible
  • Ensure subscriptions are properly set up
  • Check ROS bridge configuration allows the topic

Issue: Performance Degradation with Many Joints

Solutions:

  • Implement message throttling
  • Use interpolation for smooth visualization
  • Optimize Unity rendering for complex humanoid models
  • Consider using lower update rates for less critical joints

Best Practices for Humanoid Integration

1. Efficient Data Transfer

  • Use appropriate message types for Unity
  • Implement message compression for high-frequency data
  • Batch updates when possible
  • Use efficient serialization formats

2. Robust Communication

  • Implement connection status monitoring
  • Add retry mechanisms for failed communications
  • Use appropriate buffer sizes
  • Handle message timeouts gracefully

3. Performance Considerations

  • Optimize Unity scenes for real-time performance
  • Use object pooling for frequently created objects
  • Implement efficient rendering techniques
  • Monitor frame rate and adjust complexity accordingly

4. Error Handling

  • Implement graceful degradation when connection is lost
  • Provide fallback behaviors for critical functions
  • Log connection and communication issues
  • Implement circuit breaker patterns for service calls

Integration Example: Complete Humanoid Setup

Here's a complete example of integrating a humanoid robot with ROS TCP Connector:

// Scripts/CompleteHumanoidIntegration.cs
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Sensor;
using RosMessageTypes.Geometry;
using RosMessageTypes.Nav;
using System.Collections.Generic;

[RequireComponent(typeof(ROSConnectionManager))]
[RequireComponent(typeof(HumanoidJointController))]
public class CompleteHumanoidIntegration : MonoBehaviour
{
    [Header("ROS Topics")]
    [SerializeField] private string jointStatesTopic = "/joint_states";
    [SerializeField] private string jointCommandTopic = "/joint_commands";
    [SerializeField] private string imuTopic = "/imu/data";
    [SerializeField] private string cmdVelTopic = "/cmd_vel";
    [SerializeField] private string tfTopic = "/tf";

    [Header("Humanoid Configuration")]
    [SerializeField] private Transform robotRoot;
    [SerializeField] private List<JointMapping> jointMappings = new List<JointMapping>();

    private ROSConnection rosConnection;
    private HumanoidJointController jointController;
    private ROSConnectionManager connectionManager;

    void Start()
    {
        // Get required components
        rosConnection = ROSConnection.GetOrCreateInstance();
        jointController = GetComponent<HumanoidJointController>();
        connectionManager = GetComponent<ROSConnectionManager>();

        // Set up joint mappings
        foreach (var mapping in jointMappings)
        {
            jointController.AddJointMapping(mapping.rosJointName, mapping.unityJoint, mapping.jointType);
        }

        // Subscribe to topics
        rosConnection.Subscribe<JointStateMsg>(jointStatesTopic, OnJointStatesReceived);
        rosConnection.Subscribe<ImuMsg>(imuTopic, OnIMUReceived);
        rosConnection.Subscribe<OdometryMsg>("/odom", OnOdometryReceived);

        Debug.Log("Complete humanoid integration initialized");
    }

    void OnJointStatesReceived(JointStateMsg msg)
    {
        var jointPositions = new Dictionary<string, double>();

        for (int i = 0; i < msg.name.Count && i < msg.position.Count; i++)
        {
            jointPositions[msg.name[i]] = msg.position[i];
        }

        jointController.UpdateRobotJoints(jointPositions);
    }

    void OnIMUReceived(IMUMessage msg)
    {
        // Handle IMU data for orientation visualization
        // Implementation depends on your specific needs
    }

    void OnOdometryReceived(OdometryMsg msg)
    {
        // Update robot position in Unity world
        Vector3 position = new Vector3(
            (float)msg.pose.pose.position.x,
            (float)msg.pose.pose.position.z,
            (float)msg.pose.pose.position.y
        );

        robotRoot.position = position;
    }

    // Method to send commands back to ROS
    public void SendJointCommands(Dictionary<string, float> commands)
    {
        var jointState = new JointStateMsg();
        jointState.header = new HeaderMsg();
        jointState.header.stamp = new TimeStamp(ROSConnection.GetNodeTimeInSecs());
        jointState.header.frame_id = "base_link";

        foreach (var cmd in commands)
        {
            jointState.name.Add(cmd.Key);
            jointState.position.Add(cmd.Value);
        }

        rosConnection.Publish(jointCommandTopic, jointState);
    }

    public void SendVelocityCommand(float linearX, float angularZ)
    {
        var twist = new TwistMsg();
        twist.linear = new Vector3Msg(linearX, 0, 0);
        twist.angular = new Vector3Msg(0, 0, angularZ);

        rosConnection.Publish(cmdVelTopic, twist);
    }
}
